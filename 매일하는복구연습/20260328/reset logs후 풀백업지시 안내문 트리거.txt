CREATE OR REPLACE TRIGGER trg_after_resetlogs_check
AFTER STARTUP ON DATABASE
DECLARE
    v_reset_time DATE;
    v_db_role    VARCHAR2(30);
BEGIN
    -- 1. 현재 DB의 역할(Primary/Standby) 확인
    v_db_role := SYS_CONTEXT('USERENV', 'DATABASE_ROLE');

    -- 2. 마지막 RESETLOGS가 발생한 시간 조회
    SELECT resetlogs_time INTO v_reset_time FROM v$database;

    -- 3. 조건 체크: 운영 서버이면서 + RESETLOGS 발생한 지 10분(10/1440) 이내인 경우
    IF v_db_role = 'PRIMARY' AND v_reset_time > (SYSDATE - 10/1440) THEN
        -- Alert Log에 강력한 경고 메시지 출력
        DBMS_SYSTEM.KSDWRT(2, '##################################################');
        DBMS_SYSTEM.KSDWRT(2, '!!! ALERT: DATABASE OPENED WITH RESETLOGS !!!');
        DBMS_SYSTEM.KSDWRT(2, '!!! ACTION: PLEASE RUN FULL BACKUP IMMEDIATELY !!!');
        DBMS_SYSTEM.KSDWRT(2, '##################################################');
    END IF;
EXCEPTION
    WHEN OTHERS THEN
        NULL; -- 트리거 때문에 DB 기동이 방해받지 않도록 예외 처리
END;
/

## 🔍 v$backup_datafile 조인이 있었던 이유 
1. "껍데기"와 "알맹이" 구분
v$backup_set (껍데기): 백업이 언제 끝났고, 번호가 무엇인지 같은 요약 정보만 있습니다.

v$backup_datafile (알맹이): 그 백업 세트 안에 실제 .dbf 파일이 들어있는지 확인해 줍니다.

조인 목적: 오라클이 지 혼자 수행하는 '컨트롤파일 자동 백업'이나 '아카이브 로그 백업'은 데이터 파일(users01.dbf 등)을 포함하지 않습니다. 그래서 이 테이블과 조인하면 **"데이터 파일이 하나라도 포함된 백업"**만 필터링할 수 있었습니다.


아래 조건들은 결국 resetlogs후에 자동으로 수행되는 백업을제외한 진짜 풀백업이 있나 없나를 확인하기 위한 조치이다.
