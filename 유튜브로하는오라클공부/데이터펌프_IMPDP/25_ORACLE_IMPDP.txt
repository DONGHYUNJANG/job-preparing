-- TDE, CUST 테이블 상태 확인
-- SQL Developer 실행(SYS 접속)
SELECT 
    OWNER,
    OBJECT_TYPE,
    OBJECT_NAME,
    CREATED,
    'DROP '||OBJECT_TYPE||' '||OWNER||'.'||OBJECT_NAME||';' AS "DROP_SCRIPT"
FROM DBA_OBJECTS
WHERE OBJECT_TYPE IN ('TABLE','INDEX')
AND OBJECT_NAME IN ('CUST','CUST_IDX1','TDE_TEST3','TDE_TEST4')
ORDER BY 1,2,3;

/**********************************************************
  OBJECT CHECK 스크립트
    리눅스 서버 접속(oracle 계정)
**********************************************************/

cd /home/oracle
vi object_check.sql

COL OWNER FOR A15;
COL SEGMENT_TYPE FOR A20;
COL SEGMENT_NAME FOR A20;
COL "SIZE(MB)" FOR 999999999;
COL USERNAME FOR A20;
COL DEFAULT_TABLESPACE FOR A20;

PROMPT <<OBJECT STATUS CHECK>>
SELECT 
    OWNER,
    SEGMENT_TYPE,
    SEGMENT_NAME,
    ROUND(BYTES/1024/1024) AS "SIZE(MB)"
FROM DBA_SEGMENTS
WHERE SEGMENT_TYPE IN ('TABLE','INDEX')
AND SEGMENT_NAME IN ('CUST','CUST_IDX1','TDE_TEST3','TDE_TEST4')
ORDER BY 1,2,3;

PROMPT <<USER STATUS CHECK>>
SELECT 
    USERNAME, 
    ACCOUNT_STATUS, 
    DEFAULT_TABLESPACE 
FROM DBA_USERS
WHERE USERNAME = 'STR';
