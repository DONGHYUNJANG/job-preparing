/**********************************************************
  DB계정 생성 & 테이블스페이스 생성(STR계정 백업 준비 작업)
    DB계정명: STR
    테이블스페이스명
       데이터 영역: TS_STR_D
       인덱스 영역: TS_STR_I
    테이블스페이스 위치: /u01/app/oracle/oradata/STR
    SQL Developer실행(SYS 접속)
**********************************************************/

-- 테이블스페이스 상태 확인
SELECT 
    TABLESPACE_NAME, 
    FILE_NAME, 
    BYTES/1024/1024 AS "SIZE(MB)" 
FROM DBA_DATA_FILES;


-- 테이블스페이스 생성
-- 테이블 영역: TS_STR_D 
CREATE TABLESPACE TS_STR_D
DATAFILE '/u01/app/oracle/oradata/ORCL/TS_STR_D_001.DBF' SIZE 2048M;

-- 인덱스 영역: TS_STR_I
CREATE TABLESPACE TS_STR_I
DATAFILE '/u01/app/oracle/oradata/ORCL/TS_STR_I_001.DBF' SIZE 2048M;


-- DB 계정 STR 생성
CREATE USER STR
IDENTIFIED BY "oracle"
DEFAULT TABLESPACE TS_STR_D;

GRANT RESOURCE, CONNECT, UNLIMITED TABLESPACE TO STR;


-- DB 계정 STR 생성 확인
SELECT
    USERNAME,
    ACCOUNT_STATUS,
    CREATED,
    DEFAULT_TABLESPACE,
    TEMPORARY_TABLESPACE
FROM DBA_USERS
WHERE USERNAME = 'STR';

REVOKE READ,WRITE ON DIRECTORY STR_DUMP FROM PUBLIC;
DROP DIRECTORY STR_DUMP;

  Datapump 전체 백업
    리눅스 서버 접속(oracle 계정)
**********************************************************/

-- 전체 백업(기본+TDE)

# ============================================================
# ?? 13. 리눅스 TOP 모니터링 단축키 (Monitoring Tips)
# ============================================================

### [단위 변경] 메모리 수치 가독성 조절
# - e (소문자): 프로세스별 RES, VIRT 메모리 단위 변경 (MiB, GiB...)
# - E (대문자): 상단 전체 메모리(KiB Mem) 합계 단위 변경

### [정렬] 리소스 소모량 순서대로 보기
# - M: 메모리(Memory) 사용량 높은 순 정렬
# - P: CPU 사용량 높은 순 정렬 (기본값)
# - T: 실행 시간(Time) 누적 순 정렬

### [기타] 유용한 화면 제어
# - 1 (숫자 1): CPU 코어별 사용량 상세 보기/숨기기
# - c: 프로세스의 전체 경로(Command line) 표시/숨기기
# - k: 프로세스 강제 종료 (PID 입력 필요)
# - q: top 종료

# ============================================================
# ? 14. 데이터 펌프 리소스 제어 (Resource Control)
# ============================================================

### [실시간] 실행 중인 작업의 병렬도(Parallel) 낮추기
1. 실행 중인 터미널에서 `Ctrl + C` 입력 (작업이 중단되지 않고 'Export>' 프롬프트가 뜹니다)
2. 병렬도 변경 명령어 입력 (예: 4개에서 2개로 줄임)
   Export> PARALLEL=2
3. 다시 작업 화면으로 돌아가기
   Export> CONTINUE_CLIENT

### [조회] 현재 진행 중인 데이터 펌프 작업 찾기
# - 터미널을 껐거나, 어떤 작업이 CPU를 먹는지 모를 때 사용
SELECT owner_name, job_name, operation, state, degree 
FROM dba_datapump_jobs;

### [중단] 작업을 완전히 멈추고 삭제하고 싶을 때
# - 단순히 Ctrl + C 후 Exit 하면 서버에선 계속 돕니다. 완전히 죽이려면:
1. `Ctrl + C` 로 관리 모드 진입
2. Export> STOP_JOB=IMMEDIATE
3. 정말 죽일 건지 물으면 'YES' 입력

### [재개] 중단된 작업 다시 시작하기 (Restart)
expdp hr/password ATTACH=job_name
Export> START_JOB


# ============================================================
# ? 15. 변경된 수치 확인
# ============================================================
### [방법 1] Interactive Mode(관리 모드) 내에서 확인
# 1. Ctrl + C 입력 후 프롬프트 대기
# 2. STATUS 명령 입력
Export> STATUS

# 결과 예시:
# Job: SYS_EXPORT_FULL_01
# Operation: EXPORT
# Mode: FULL
# State: EXECUTING
# Parallel: 2  <-- 여기서 변경된 수치를 확인할 수 있습니다.

### [방법 2] 별도 SQL 터미널에서 딕셔너리 뷰 조회
# - 작업 중인 창을 건드리지 않고 외부에서 감시할 때 유용합니다.
COL job_name FOR A20
COL state    FOR A15
SELECT job_name, operation, state, degree 
FROM dba_datapump_jobs;

# * 여기서 'DEGREE' 컬럼의 숫자가 현재 작동 중인 병렬도(Parallel)입니다.
