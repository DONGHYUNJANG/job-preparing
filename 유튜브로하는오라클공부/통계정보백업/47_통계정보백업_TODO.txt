------------------------------------------------------------------
-- 리눅스 디렉터리 생성
-- oracle 계정에서 수행
------------------------------------------------------------------
## Shell 스크립트 디렉터리 생성
mkdir /home/oracle/script

## Shell 로그 백업 디렉터리 생성
mkdir /home/oracle/oraclelog



------------------------------------------------------------------
-- 자동 백업 스크립트 생성
-- oracle 계정에서 수행
------------------------------------------------------------------
cat << \FILE > /home/oracle/script/dbst_str_backup.sh
#!/bin/bash

################################################################################
#  STATISTICS BACKUP SCRIPT
#    V 1.0    통계정보 백업  DBBODA
#             백업 대상 계정: STR
################################################################################

export ORACLE_HOME=/oracle/app/oracle/product/19.0.0.0/db_1
export ORACLE_SID=STR
export PATH=$ORACLE_HOME/bin:$PATH
export DATE=`date +%Y%m%d`
export LOG=/home/oracle/oraclelog/dbst_STR_BACKUP_${DATE}.log

sqlplus / as sysdba << EOF >> ${LOG}

EXEC DBMS_STATS.CREATE_STAT_TABLE(OWNNAME=>'SYSTEM', STATTAB=>'DBST_BACKUP_STR_${DATE}', TBLSPACE=>'SYSTEM');

BEGIN
 FOR X IN (
          SELECT
            TABLE_NAME
          FROM DBA_TABLES
          WHERE OWNER = 'STR'
          )
 LOOP
    DBMS_STATS.EXPORT_TABLE_STATS(OWNNAME=>'STR',TABNAME=>X.TABLE_NAME,STATTAB=>'DBST_BACKUP_STR_${DATE}',CASCADE=>TRUE,STATOWN=>'SYSTEM');
 END LOOP;
END;
/

EXIT
EOF

FILE


------------------------------------------------------------------
-- Shell 스크립트 권한 변경
-- oracle 계정에서 수행
------------------------------------------------------------------
## Shell 스크립트 권한 변경
chmod 755 /home/oracle/script/dbst_str_backup.sh



------------------------------------------------------------------
-- crontab 생성
-- oracle 계정에서 수행
------------------------------------------------------------------
crontab -e

## ORACLE GATHER STATS BACKUP (STR)
10 00 * * * /home/oracle/script/dbst_str_backup.sh



------------------------------------------------------------------
-- 데이터 확인
-- sqlplus / as sysdba 수행
------------------------------------------------------------------

sqlplus -s / as sysdba << EOF
SELECT
    DISTINCT C1
FROM SYSTEM.DBST_BACKUP_STR_20230404
ORDER BY C1;
EXIT
EOF
