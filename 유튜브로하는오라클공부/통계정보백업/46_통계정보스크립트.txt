----------------------------------------------------------------------------------------------------------------------------------------------------------------------------
-- 테이블 통계정보 확인
--  SYS 계정에서 작업
--  DB 계정: STR
--  테이블명: CUST, CUST_DTL
----------------------------------------------------------------------------------------------------------------------------------------------------------------------------
SELECT 
    OWNER,
    TABLE_NAME,
    TO_CHAR(LAST_ANALYZED,'YYYY/MM/DD HH24:MI:SS') AS "LAST_ANALYZED"
FROM DBA_TABLES
WHERE OWNER = 'STR';




----------------------------------------------------------------------------------------------------------------------------------------------------------------------------
-- 통계정보 백업 테이블 생성
----------------------------------------------------------------------------------------------------------------------------------------------------------------------------
EXEC DBMS_STATS.CREATE_STAT_TABLE(OWNNAME=>'SYSTEM', STATTAB=>'DBST_BACKUP_STR_20230321', TBLSPACE=>'SYSTEM');




----------------------------------------------------------------------------------------------------------------------------------------------------------------------------
-- 통계정보 백업
----------------------------------------------------------------------------------------------------------------------------------------------------------------------------
SELECT
    OWNER,
    TABLE_NAME,
    'EXEC DBMS_STATS.EXPORT_TABLE_STATS(OWNNAME=>'''||OWNER||''', TABNAME=>''' ||
    TABLE_NAME|| ''', STATTAB=>''DBST_BACKUP_STR_20230321'''||
    ', CASCADE=>TRUE, STATOWN=>''SYSTEM'');' AS "SCRIPT"
FROM DBA_TABLES
WHERE OWNER = 'STR'
ORDER BY 1, 2;

-- 통계정보 백업 확인
SELECT * FROM SYSTEM.DBST_BACKUP_STR_20230321;




----------------------------------------------------------------------------------------------------------------------------------------------------------------------------
-- 통계정보 갱신
----------------------------------------------------------------------------------------------------------------------------------------------------------------------------
SELECT
    OWNER,
    TABLE_NAME,
    'EXEC DBMS_STATS.GATHER_TABLE_STATS(OWNNAME=>'''||OWNER||''', TABNAME=>'''||
    TABLE_NAME||''', METHOD_OPT=>''FOR ALL COLUMNS SIZE 1'', DEGREE =>4);' AS "SCRIPT"
FROM DBA_TABLES
WHERE OWNER = 'STR'
ORDER BY 1, 2;


----------------------------------------------------------------------------------------------------------------------------------------------------------------------------
-- 통계정보 백업 테이블을 이용한 원복
----------------------------------------------------------------------------------------------------------------------------------------------------------------------------
SELECT
    OWNER,
    TABLE_NAME,
	'EXEC DBMS_STATS.IMPORT_TABLE_STATS(OWNNAME=>'''||OWNER||''', TABNAME=>'''||TABLE_NAME||''''||
    ', STATTAB=>''DBST_BACKUP_STR_20230321'', CASCADE=>TRUE, STATOWN=>''SYSTEM'', NO_INVALIDATE=> FALSE);' AS SCRIPT
FROM DBA_TABLES
WHERE OWNER = 'STR'
ORDER BY 1, 2;




----------------------------------------------------------------------------------------------------------------------------------------------------------------------------
-- 테이블 통계정보 LOCK, UNLOCK
----------------------------------------------------------------------------------------------------------------------------------------------------------------------------

-- 테이블 통계정보 자동 갱신(X) LOCK
EXEC DBMS_STATS.LOCK_TABLE_STATS('STR','CUST');

-- 테이블 통계정보 자동 갱신(O) UNLOCK
EXEC DBMS_STATS.UNLOCK_TABLE_STATS('STR','CUST');

EXEC DBMS_STATS.GATHER_TABLE_STATS(OWNNAME=>'STR', TABNAME=>'CUST', METHOD_OPT=>'FOR ALL COLUMNS SIZE 1');
