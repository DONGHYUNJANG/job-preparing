# 컬럼별 너비 및 헤더 설정
COL file_id         FOR 999          HEAD 'ID'
COL tablespace_name FOR A20          HEAD 'Tablespace'
COL file_name       FOR A60          HEAD 'Physical_Path'
COL "Size(MB)"      FOR 9,999,999    HEAD 'Size(MB)'
COL "MaxSize(MB)"   FOR 9,999,999    HEAD 'Max(MB)'
COL "Auto"          FOR A5           HEAD 'Auto'
COL "Total(MB)"     FOR 9,999,999    HEAD 'Total(MB)'
COL "Used(MB)"      FOR 9,999,999    HEAD 'Used(MB)'
COL "Used(%)"       FOR 999.99       HEAD 'Used(%)'

### [조회 1] 모든 데이터파일의 실제 경로 및 용량
SELECT 
    file_id,
    tablespace_name,
    file_name,
    bytes/1024/1024 AS "Size(MB)",
    autoextensible AS "Auto"
FROM dba_data_files
ORDER BY tablespace_name, file_id;

### [조회 2] 임시(Temp) 파일 경로 확인
SELECT 
    file_id,
    tablespace_name,
    file_name,
    bytes/1024/1024 AS "Size(MB)"
FROM dba_temp_files;

### [조회 3] 테이블스페이스별 현재 사용량 (%)
SELECT 
    df.tablespace_name,
    ROUND(df.bytes / (1024 * 1024)) AS "Total(MB)",
    ROUND((df.bytes - NVL(SUM(fs.bytes),0)) / (1024 * 1024)) AS "Used(MB)",
    ROUND((df.bytes - NVL(SUM(fs.bytes),0)) / df.bytes * 100, 2) AS "Used(%)"
FROM dba_data_files df
LEFT JOIN dba_free_space fs ON df.file_id = fs.file_id
GROUP BY df.tablespace_name, df.bytes
ORDER BY 4 DESC;

# 📦 10. Data Pump 실행 옵션 (Practical Options)
# ============================================================

### [EXPORT] 자주 쓰는 내보내기 옵션 (expdp)
# - SCHEMAS: 특정 유저 전체 백업
# - PARALLEL: 프로세스 병렬 처리 (속도 향상)
# - COMPRESSION: 덤프 파일 압축 (용량 절감)
expdp hr/password@orcl \
DIRECTORY=dpump_dir \
DUMPFILE=hr_full_%U.dmp \
LOGFILE=exp_hr.log \
SCHEMAS=hr \
PARALLEL=4 \
COMPRESSION=ALL;

### [IMPORT] 자주 쓰는 들여오기 옵션 (impdp)
# - REMAP_SCHEMA: A유저 데이터를 B유저로 옮길 때
# - REMAP_TABLESPACE: 소스 테이블스페이스가 타겟에 없을 때
# - TABLE_EXISTS_ACTION: 테이블이 이미 있을 때 처리 (APPEND, REPLACE, TRUNCATE)
impdp system/password@orcl \
DIRECTORY=dpump_dir \
DUMPFILE=hr_full_%U.dmp \
LOGFILE=imp_hr.log \
PARALLEL=4 \
REMAP_SCHEMA=hr:scott \
REMAP_TABLESPACE=users:ts_data \
TABLE_EXISTS_ACTION=REPLACE;

### [FILTER] 특정 조건만 골라내기 (Include/Exclude)
# - EXCLUDE: 특정 오브젝트 제외 (예: 통계정보, 인덱스 제외)
# - CONTENT: 데이터만(DATA_ONLY) 또는 구조만(METADATA_ONLY)
expdp hr/password@orcl \
DIRECTORY=dpump_dir \
DUMPFILE=hr_meta.dmp \
CONTENT=METADATA_ONLY \
EXCLUDE=STATISTICS,INDEX;

### [권한] DB Link 생성을 위한 권한 부여 (System 계정에서 실행)
GRANT CREATE DATABASE LINK TO hr;
# (Public 생성 시) GRANT CREATE PUBLIC DATABASE LINK TO hr;

### [생성] 방법 1. TNS 정보를 직접 입력하는 방식 (권장)
# - tnsnames.ora 설정을 건드리지 않아도 되어 간편합니다.
CREATE DATABASE LINK remote_link
CONNECT TO remote_user IDENTIFIED BY "password123!"
USING '(DESCRIPTION=(ADDRESS=(PROTOCOL=TCP)(HOST=192.168.0.100)(PORT=1521))(CONNECT_DATA=(SID=orcl)))';

### [생성] 방법 2. TNS 명칭을 사용하는 방식
# - 로컬 서버의 tnsnames.ora에 'PROD_DB'가 정의되어 있어야 합니다.
CREATE DATABASE LINK prod_link
CONNECT TO scott IDENTIFIED BY tiger
USING 'PROD_DB';

### [조회] 생성된 DB Link 목록 확인
COL owner FOR A10
COL db_link FOR A20
COL host FOR A30
SELECT owner, db_link, username, host FROM dba_db_links;

### [사용] 원격지 테이블 조회 및 데이터 복사
# - 테이블명 뒤에 '@링크이름'을 붙입니다.
SELECT * FROM emp@remote_link;

# - 원격지 데이터를 로컬로 바로 복사 (CTAS)
CREATE TABLE local_emp_bak AS SELECT * FROM emp@remote_link;

### [삭제] DB Link 제거
DROP DATABASE LINK remote_link;
# (Public 삭제 시) DROP PUBLIC DATABASE LINK remote_link;

# ⚡ 16. 이미지 카피 - 데이터파일 단위
# ============================================================
RUN {
  # 1. 병렬 처리를 위해 채널 할당 (I/O 분산)
  ALLOCATE CHANNEL c1 DEVICE TYPE DISK;
  ALLOCATE CHANNEL c2 DEVICE TYPE DISK;

  # 2. 이미지 카피 생성 (핵심 옵션 조합)
  BACKUP AS COPY 
    DATAFILE 12                                      -- 대상: 데이터파일 12번 (ts700)
    FORMAT '/u01/app/oracle/oradata/ORCL/backup/df12_%U.copy' -- 저장 경로 및 포맷
    TAG 'IMG_COPY_TS700_LAB'                         -- 식별용 태그 이름
    CHECK LOGICAL                                    -- 내부 논리적 손상까지 검사
    SECTION SIZE 100M;                               -- 대용량 파일 병렬 카피 (멀티섹션)

  # 3. 생성된 카피본 정보 확인
  LIST COPY OF DATAFILE 12;

  RELEASE CHANNEL c1;
  RELEASE CHANNEL c2;
}

# ============================================================
## 💡 개념 설명: ASMCMD (ASM Command-Line Utility)
# ============================================================


**ASMCMD**는 Oracle ASM(Automatic Storage Management) 인스턴스를 관리하기 위한 명령줄 툴입니다. SQL*Plus에서 SQL 문으로 ASM을 관리할 수도 있지만, `asmcmd`는 마치 리눅스의 셸(Shell)처럼 파일 시스템을 탐색하고 관리할 수 있게 해줍니다.

데이터베이스 엔지니어(DBA)가 ASM 디스크 그룹 내의 파일 구조를 확인하거나, 디렉토리를 생성하고, 디스크 상태를 점검할 때 가장 빈번하게 사용합니다.


## 🛠 자주 사용되는 ASMCMD 명령어

`asmcmd`를 실행한 후 사용할 수 있는 주요 명령어들을 기능별로 정리해 드립니다.

### 1. 파일 및 디렉토리 관리 (Linux 유사 명령어)
ASM 내부를 일반 파일 시스템처럼 탐색할 때 사용합니다.
* `ls`: 디렉토리나 파일 목록을 확인합니다. (옵션 `-l`로 상세 정보 확인 가능)
* `cd`: 디렉토리를 이동합니다. (예: `cd +DATA/ORCL/CONTROLFILE`)
* `pwd`: 현재 위치한 ASM 경로를 보여줍니다.
* `mkdir`: 새로운 디렉토리를 생성합니다.
* `rm`: 파일이나 빈 디렉토리를 삭제합니다. (주의: `-r` 옵션은 하위 항목까지 모두 삭제)

### 2. 디스크 그룹 관리 및 상태 확인
ASM의 핵심인 디스크 그룹의 상태를 파악할 때 필수적입니다.
* `lsdg`: 모든 디스크 그룹의 이름, 상태, 총 용량, 사용량을 확인합니다. (가장 많이 사용됨)
* `lsattr`: 디스크 그룹의 속성(Attribute)을 확인합니다.
* `chdg`: 디스크 그룹의 설정을 변경합니다.
* `mount` / `umount`: 특정 디스크 그룹을 마운트하거나 해제합니다.

### 3. 디스크 및 장치 정보
물리적인 디스크 상태를 확인할 때 사용합니다.
* `lsdsk`: ASM 디스크의 경로, 상태, 용량 등을 나열합니다. (옵션 `-p`나 `-k`를 주로 사용)
* `iostat`: ASM 디스크의 I/O 통계를 실시간으로 확인합니다.

### 4. 기타 유용한 명령어
* `du`: 특정 디렉토리가 점유하고 있는 실제 용량을 확인합니다.
* `cp`: ASM과 일반 OS 파일 시스템 간에 파일을 복사합니다. (예: `cp +DATA/temp.dbf /home/oracle/`)
* `help`: 특정 명령어의 사용법이 기억나지 않을 때 `help [명령어]` 형식으로 사용합니다.

---

### ⚠️ 실무 팁
* `asmcmd`를 실행하기 전에 반드시 **Grid Infrastructure 환경 변수**(`ORACLE_HOME` 및 `ORACLE_SID`)가 ASM 인스턴스에 맞춰 설정되어 있어야 합니다.
* 일반적으로 `ORACLE_SID`는 `+ASM` 또는 `+ASM1`(RAC의 경우)입니다.

## 💡 FRA (Fast Recovery Area)와 ASM 경로
# ============================================================

현재 설정된 db_recovery_file_dest의 값인 +DATA는 일반 OS 디렉토리가 아니라 Oracle ASM(Automatic Storage Management) 내의 디스크 그룹(Disk Group) 이름입니다.

Oracle은 FRA(Fast Recovery Area)를 ASM에 저장할 때, 사용자가 일일이 경로를 지정하지 않아도 OMF(Oracle Managed Files) 규칙에 따라 자동으로 구조화된 디렉토리를 생성하여 파일을 관리합니다. 따라서 단순히 +DATA라고만 표시되어도 그 하위에 정해진 규칙대로 파일들이 숨어 있습니다.

🛠 문제 해설: 실제 위치 찾는 방법
+DATA 내에서 아카이브 로그나 백업 파일이 정확히 어디 있는지 확인하려면 ASMCMD 툴을 사용하거나 V$ 뷰를 조회해야 합니다.

1. ASMCMD를 이용한 탐색 (가장 직관적인 방법)
터미널에서 asmcmd에 접속하여 리눅스처럼 디렉토리를 타고 들어가 봅니다. 보통 아래와 같은 구조로 생성됩니다.
+데이터그룹명/DB이름/ARCHIVELOG/날짜/파일명

Bash
# 1. ASMCMD 접속 (Grid 유저 또는 oracle 유저)
asmcmd

# 2. 경로 이동 (보통 DB명이 하위 폴더로 존재합니다)
ASMCMD> cd +DATA/ORCL/ARCHIVELOG
ASMCMD> ls -al
위에서 ORCL은 본인의 DB_UNIQUE_NAME입니다.

2. SQL을 이용한 확인 (현재 아카이브 로그 위치 조회)
현재 실제로 기록되고 있는 아카이브 파일들의 전체 경로를 SQL*Plus에서 바로 확인할 수 있습니다.

SQL
SELECT name FROM v$archived_log WHERE completion_time > sysdate - 1;
이 쿼리를 실행하면 +DATA/ORCL/ARCHIVELOG/2026_04_01/thread_1_seq_16... 와 같은 실제 풀 경로가 출력됩니다.

3. 디스크 그룹 내 사용량 확인
+DATA 그룹 내에서 FRA가 얼마나 공간을 차지하고 있는지 확인하려면 다음 뷰를 조회하세요.

SQL
SELECT * FROM v$recovery_area_usage;

# 특정 테이블에만 SPPLEMENTAL LOG 설정

① 특정 컬럼들만 로그에 남기기 (가장 효율적)
모든 컬럼이 아니라, 분석에 꼭 필요한 컬럼(예: ID, 이름)만 지정해서 로그를 남깁니다.

SQL
ALTER TABLE SCOTT.EMP ADD SUPPLEMENTAL LOG GROUP log_grp_emp (EMPNO, ENAME) ALWAYS;
② 해당 테이블의 모든 컬럼 남기기 (PK 없을 때 추천)
오늘 사용자님을 괴롭혔던 PK 없는 테이블은 이 명령어로 해당 테이블만 '무적'으로 만들면 됩니다.

SQL
ALTER TABLE SCOTT.EMP ADD SUPPLEMENTAL LOG DATA (ALL) COLUMNS;
주의: DATABASE LEVEL 설정과 문법이 거의 비슷하지만, 앞에 TABLE명이 들어갑니다.

③ PK/Unique/Foreign Key 기반 설정
SQL
-- 테이블의 PK 컬럼들을 로그에 남김
ALTER TABLE SCOTT.EMP ADD SUPPLEMENTAL LOG DATA (PRIMARY KEY) COLUMNS;

-- 테이블의 Unique Index 컬럼들을 로그에 남김
ALTER TABLE SCOTT.EMP ADD SUPPLEMENTAL LOG DATA (UNIQUE) COLUMNS;

2. 설정 확인하는 방법 (딕셔너리 뷰)
설정을 마친 후에는 V$DATABASE가 아니라 ALL_LOG_GROUPS 또는 ALL_LOG_GROUP_COLUMNS 뷰를 조회해야 합니다.

SQL
-- 어떤 테이블에 어떤 로그 그룹이 설정되어 있는지 확인
SELECT OWNER, TABLE_NAME, LOG_GROUP_NAME, LOG_GROUP_TYPE
FROM ALL_LOG_GROUPS

-- 로그 그룹에 포함된 상세 컬럼 확인

FROM ALL_LOG_GROUP_COLUMNS
WHERE TABLE_NAME = 'EMP';

3. 설정 삭제하는 방법 (원복)
테스트가 끝났거나 부하가 너무 심해져서 설정을 지우고 싶을 때는 DROP을 사용합니다.

SQL
-- 특정 로그 그룹 삭제
ALTER TABLE SCOTT.EMP DROP SUPPLEMENTAL LOG GROUP log_grp_emp;

-- ALL COLUMNS 등 데이터 레벨 옵션 삭제
ALTER TABLE SCOTT.EMP DROP SUPPLEMENTAL LOG DATA (ALL) COLUMNS;




PK가 없는 테이블의 DELETE 이력을 보려면 아래 명령어를 실행해야 합니다.

SQL

# 🗂️ Oracle PDB(Pluggable Database) 오픈 명령어 가이드
# =================================================

---

## 💡 1. 핵심 개념 설명

Oracle 12c부터 도입된 **Multitenant Architecture**에서 PDB는 CDB(Container Database) 내에 존재하는 독립적인 데이터베이스입니다. 

* **기본 동작:** CDB가 시작(Startup)될 때, 내부에 포함된 PDB들은 기본적으로 **`MOUNT`** 상태(닫힌 상태)로 시작됩니다.
* **관리자 작업:** 사용자가 PDB에 접속하여 데이터를 조회하거나 수정하려면, 관리자가 직접 PDB를 **`READ WRITE`** 모드로 열어주어야 합니다.

---

## 🛠️ 2. PDB 오픈 명령어 정리

모든 명령어는 `SYSDBA` 권한이 있는 상태에서 실행해야 합니다.

### A. 특정 PDB 열기
가장 일반적으로 사용되는 명령어로, CDB(Root)에서 특정 PDB 하나만 지정해서 엽니다.
```sql
-- 문법: ALTER PLUGGABLE DATABASE [PDB_NAME] OPEN;
ALTER PLUGGABLE DATABASE pdb1 OPEN;
```

### B. 모든 PDB 한꺼번에 열기
CDB 내의 모든 PDB를 동시에 오픈할 때 사용합니다.
```sql
ALTER PLUGGABLE DATABASE ALL OPEN;
```

### C. 특정 PDB를 제외하고 모두 열기
```sql
ALTER PLUGGABLE DATABASE ALL EXCEPT pdb2 OPEN;
```

### D. 읽기 전용(Read Only)으로 열기
```sql
ALTER PLUGGABLE DATABASE pdb1 OPEN READ ONLY;
```

---

## 📝 PDB 상태 저장

### ✅ 현재 상태 확인 (v$pdbs)
명령어 실행 후 PDB의 상태가 `READ WRITE`인지 확인하는 습관이 중요합니다.
```sql
SELECT name, open_mode FROM v$pdbs;
```

### ✅ PDB 상태 저장 (`SAVE STATE`)
CDB를 재시작해도 PDB가 자동으로 열리게 하려면 현재의 오픈 상태를 저장해야 합니다. (OCP 단골 문제!)
```sql
-- 특정 PDB의 현재 상태 저장
ALTER PLUGGABLE DATABASE pdb1 SAVE STATE;

-- 모든 PDB의 현재 상태 저장
ALTER PLUGGABLE DATABASE ALL SAVE STATE;

-- 저장된 상태 해제 (다시 수동으로 열어야 함)
ALTER PLUGGABLE DATABASE pdb1 DISCARD STATE;
```

# 🗂️ RAC환경 로그파일 삭제 및 추가
# =================================================

---

## 💡 1. 핵심 개념 설명: Thread와 공유 저장소

RAC 환경에서 가장 중요한 점은 **"로그는 각자 쓰고, 읽는 건 같이 한다"**는 것입니다.

* **Thread의 분리**: 각 인스턴스(Node)는 충돌을 방지하기 위해 자신만의 전용 로그 쓰레드(Thread)를 가집니다.
* **저장소의 공유**: 하지만 장애 복구 시 다른 노드가 이 로그를 읽어야 하므로, 모든 Redo Log 파일은 **ASM(+DATA 등)**과 같은 공유 저장소에 위치해야 합니다.

---

## 🛠️ 2. RAC 로그 관리 명령어 

### ✅ 로그 상태 확인 (전체 노드)
어떤 인스턴스가 어떤 쓰레드를 쓰고 있는지, 로그 파일이 어디 있는지 한눈에 확인하는 쿼리입니다.
```sql
SELECT a.thread#, a.group#, b.member, a.bytes/1024/1024 as "SIZE(MB)", a.status
FROM v$log a, v$logfile b
WHERE a.group# = b.group#
ORDER BY a.thread#, a.group#;
```

### ✅ 로그 그룹 추가 (Thread 지정 필수)
RAC에서는 단순히 추가하는 것이 아니라, **어느 인스턴스(Thread)가 사용할 것인지** 명시해야 합니다.
SIZE 생략 가능
```sql
-- Thread 1(1번 노드)에 4번 그룹 추가
ALTER DATABASE ADD LOGFILE THREAD 1 GROUP 4 '+DATA' SIZE 100M;

-- Thread 2(2번 노드)에 5번 그룹 추가
ALTER DATABASE ADD LOGFILE THREAD 2 GROUP 5 '+DATA' SIZE 100M;
```

### ✅ 로그 그룹 삭제
!!!!!!!!!!!!!!!!!!!! 삭제 전에는 반드시 해당 그룹의 상태가 **`INACTIVE`**여야 합니다. !!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
만약 `CURRENT`나 `ACTIVE`라면 위에서 배운 `SWITCH LOGFILE`이나 `CHECKPOINT`를 통해 상태를 변경해야 합니다.
```sql
-- 로그 그룹 삭제 (물리적 파일은 ASM에서 별도 삭제 필요할 수 있음)
ALTER DATABASE DROP LOGFILE GROUP 4;
```

---

## ⚠️ 주의사항 요약
1.  **최소 그룹 수**: 각 Thread는 최소 **2개 이상의 로그 그룹**을 가져야 합니다.
2.  **크기 통일**: RAC 내의 모든 리두 로그 파일은 성능과 관리 편의를 위해 **동일한 크기**로 맞추는 것이 정석입니다.
3.  **OMF 사용**: ASM 환경이라면 파일 경로를 생략하고 `'+DATA'`만 지정하여 오라클이 이름을 자동으로 짓게(OMF) 하는 것이 편리합니다.

# 🗂️ RAC 인스턴스 및 데이터베이스 재기동 가이드

---

## 💡 1. 핵심 개념 설명: `srvctl` vs `sqlplus`

* **`srvctl` (권장):** 오라클 클러스터 관리 도구입니다. 이 명령어로 DB를 내리면 리스너와의 관계, 의존성 등을 모두 고려하여 깔끔하게 처리하며, OCR(장부)에 상태를 정확히 업데이트합니다.
* **`sqlplus`:** 개별 인스턴스만 제어합니다. RAC에서는 한쪽을 내려도 다른 쪽이 살아있으므로 큰 문제는 없으나, 클러스터 관리자가 "어? 인스턴스가 왜 죽었지?" 하고 다시 살리려고 시도하는 등 간섭이 생길 수 있습니다.

---

## 🛠️ 2. 재기동 명령어 (MD 파일 정리용)

가장 많이 쓰이는 **`srvctl`** 기준으로 정리해 드립니다. (명령어는 OS 프롬프트에서 실행합니다.)

### ✅ 방법 1: 전체 데이터베이스 한 번에 내리고 올리기 (가장 간편)
양쪽 노드(인스턴스)를 한꺼번에 제어할 때 사용합니다.
```bash
# 전체 DB 중지 (Immediate 옵션)
srvctl stop database -d racdb -o immediate

# 전체 DB 시작
srvctl start database -d racdb
```

### ✅ 방법 2: 특정 인스턴스만 골라서 내리고 올리기
1번 노드만 점검하거나 재기동하고 싶을 때 사용합니다.
```bash
# 1번 인스턴스만 중지
srvctl stop instance -d racdb -i racdb1 -o immediate

# 1번 인스턴스만 시작
srvctl start instance -d racdb -i racdb1
```

### ✅ 상태 확인 (재기동 후 필수)
```bash
# 전체 인스턴스 및 서비스 상태 확인
srvctl status database -d racdb
```



---

## 📝 3. 문제 해설: 주의사항 및 OCP 포인트

> **Q: `shutdown abort`와 `immediate`의 차이는?**
> **A:** `immediate`는 현재 진행 중인 트랜잭션을 롤백하고 정상적으로 체크포인트를 수행하며 내리지만, `abort`는 전원을 끄는 것과 같아서 나중에 올릴 때 **인스턴스 복구(Instance Recovery)** 과정이 필요합니다. 실습 중이라도 가급적 `-o immediate`를 쓰시는 것이 좋습니다.

> **💡 팁:** `srvctl`로 내리면 아까 질문하신 **OCR** 정보와 동기화되기 때문에, 서버가 재부팅되어도 그리드 인프라가 자동으로 DB를 다시 올려줍니다.

---

## ✅ 요약 테이블

| 대상 | 명령어 (중지) | 명령어 (시작) |
| :--- | :--- | :--- |
| **전체 DB** | `srvctl stop database -d [DB명]` | `srvctl start database -d [DB명]` |
| **특정 노드** | `srvctl stop instance -d [DB명] -i [인스턴스명]` | `srvctl start instance -d [DB명] -i [인스턴스명]` |

요청하신 **`crsctl`** 명령어 핵심 요약 정리해 드립니다. OCP 시험과 실무에서 가장 자주 쓰이는 것들 위주로 Notion 스타일로 정리했으니 복사해서 보관하세요!

# ================================================
## 💡 Oracle Clusterware Control (crsctl) 명령어 가이드
# ================================================


`crsctl`은 오라클 클러스터웨어(Grid Infrastructure) 자체를 관리하는 강력한 도구입니다. 리소스 개별 관리(`srvctl`)보다 더 하위 계층을 다룹니다.

### 1. 클러스터 상태 확인 (Check)
가장 기본이 되는 상태 점검 명령어입니다.

| 명령어 | 설명 |
| :--- | :--- |
| `crsctl check crs` | 현재 노드의 OHAS, CSS, EVM, CRS 데몬 상태 확인 |
| `crsctl check cluster [-all]` | 로컬 또는 전체 노드의 클러스터 상태 확인 |
| `crsctl query css votedisk` | Voting Disk의 위치와 상태 확인 |

### 2. 리소스 상태 조회 (Stat)
현재 어떤 프로세스가 어디서 실행 중인지 상세히 보여줍니다.

| 명령어 | 설명 |
| :--- | :--- |
| `crsctl stat res -t` | 전체 리소스의 상태를 테이블 형식으로 요약 조회 |
| `crsctl stat res -t -init` | GI 기동 초기 단계(ohasd 수준)의 리소스 상태 확인 |
| `crsctl stat res [Resource_Name] -p` | 특정 리소스의 상세 프로필(설정값) 확인 |

### 3. 클러스터 기동 및 종료 (Start/Stop)
**반드시 `root` 계정**에서 실행해야 하는 경우가 많습니다.

| 명령어 | 설명 |
| :--- | :--- |
| `crsctl start crs` | 현재 노드의 클러스터 서비스를 시작 (ohasd 기동) |
| `crsctl stop crs [-f]` | 현재 노드의 클러스터 서비스를 종료 (-f는 강제 종료) |
| `crsctl start cluster -all` | 모든 노드의 클러스터 서비스를 한꺼번에 기동 |
| `crsctl stop cluster -all` | 모든 노드의 클러스터 서비스를 한꺼번에 종료 |

### 4. 자동 기동 설정 (Config)
서버 리부팅 시 GI가 자동으로 뜰지 말지를 결정합니다.

| 명령어 | 설명 |
| :--- | :--- |
| `crsctl enable crs` | OS 부팅 시 Clusterware 자동 기동 활성화 |
| `crsctl disable crs` | OS 부팅 시 Clusterware 자동 기동 비활성화 (패치 작업 시 필수) |

---


`srvctl`에서 인스턴스와 데이터베이스를 제어하는 명령어들은 실무뿐만 아니라 OCP 시험에서도 단골로 출제되는 핵심 영역입니다. 요청하신 인스턴스 정지 명령어를 포함하여, 자주 사용되는 **제어 관련 명령어**들을 Notion 스타일로 깔끔하게 정리해 드립니다.

---

# ================================================
## 🛠️ SRVCTL 제어 명령어 정리 (Instance & Database)
# ================================================


`srvctl`은 RAC 환경에서 개별 인스턴스나 데이터베이스 전체를 관리할 때 사용합니다.

### 1. 인스턴스(Instance) 제어
특정 노드의 인스턴스만 끄거나 켤 때 사용합니다.

| 작업 내용 | 명령어 |
| :--- | :--- |
| **인스턴스 기동** | `srvctl start instance -db racdb -instance racdb1,racdb2` |
| **인스턴스 정지** | `srvctl stop instance -db racdb -instance racdb1 -stopoption immediate` |
| **상태 확인** | `srvctl status instance -db racdb -instance racdb1` |

> **Tip:** `-stopoption`은 `normal`, `transactional`, `immediate`, `abort`를 사용할 수 있으며, 생략 시 기본값은 `immediate`입니다.

---

### 2. 데이터베이스(Database) 전체 제어
모든 노드의 인스턴스를 한꺼번에 관리할 때 사용합니다.

| 작업 내용 | 명령어 |
| :--- | :--- |
| **DB 전체 기동** | `srvctl start database -db racdb` |
| **DB 전체 정지** | `srvctl stop database -db racdb -stopoption immediate` |
| **DB 전체 상태** | `srvctl status database -db racdb` |
| **DB 설정 확인** | `srvctl config database -db racdb` |

---

### 3. 기타 주요 리소스 제어 (Listener & ASM)
DB 외에도 RAC를 구성하는 핵심 리소스들입니다.

| 리소스 | 명령어 |
| :--- | :--- |
| **Listener 확인** | `srvctl status listener` |
| **Listener 정지** | `srvctl stop listener -listener listener_name` |
| **ASM 상태 확인** | `srvctl status asm` |

---

# ================================================
### **전체 종료 및 시작 절차 (추천 방법)**
# ================================================

이 작업은 반드시 **root 계정**에서 수행해야 클러스터의 모든 데몬(ohasd, crsd 등)을 완전히 제어할 수 있습니다.

#### **Step 1: 모든 노드 전체 종료**
현재 꼬여있는 상태를 무시하고 강제로 종료하려면 `-f` (force) 옵션을 사용합니다.
```bash
# root 계정으로 수행
/u01/app/grid/19c/bin/crsctl stop cluster -all -f
```
* 만약 특정 노드(예: rac1)만 완전히 껐다 켜고 싶다면: `crsctl stop crs -f`

#### **Step 2: 프로세스 완전 종료 확인**
명령어 수행 후 오라클 관련 프로세스가 정말 다 죽었는지 확인합니다.
```bash
ps -ef | grep pmon
ps -ef | grep d.bin
```

#### **Step 3: 전체 다시 시작**
이제 깨끗해진 상태에서 클러스터 전체를 기동합니다.
```bash
# root 계정으로 수행
/u01/app/grid/19c/bin/crsctl start cluster -all
```
* 특정 노드만 켤 경우: `crsctl start crs`

---

### **2. 왜 이 방법이 좋은가요?**

* **의존성 자동 해결:** `crsctl start cluster`를 실행하면 **Storage → ASM → Listener → Database** 순서로 의존성에 맞춰서 GI가 알아서 다 올려줍니다.
* **상태 동기화:** 꼬여있던 OCR의 리소스 상태와 실제 OS의 프로세스 상태가 다시 **ONLINE**으로 동기화됩니다.
* **oratab 적용:** 이제 `oratab`을 **`Y`**로 고치셨으니, 이 명령 한 번이면 마지막 단계에서 DB까지 자동으로 `Open` 될 것입니다.

## 1. 시스템 전체의 '고통 지수' 확인 (`V$SYSTEM_EVENT`)
현재 인스턴스가 켜진 이후, 시스템 전체에서 어떤 대기 이벤트가 가장 많은 시간을 잡아먹었는지 확인합니다.

```sql
SELECT * FROM (
    SELECT 
        event,                -- 대기 이벤트 명
        total_waits,          -- 발생 횟수
        time_waited_micro,    -- 총 대기 시간 (마이크로초)
        average_wait          -- 평균 대기 시간 (한 번 기다릴 때 걸리는 시간)
    FROM v$system_event
    WHERE wait_class <> 'Idle' -- 성능과 무관한 '대기(Idle)' 상태는 제외
    ORDER BY time_waited_micro DESC -- 가장 오래 기다린 순서로 정렬
) WHERE ROWNUM <= 5; -- 상위 5개만 추출
```

---

## 2. 최근 5분간의 '현장 사진' 분석 (`V$ACTIVE_SESSION_HISTORY`)
ASH는 1초마다 샘플링한 데이터입니다. 최근 5분 동안 어떤 세션이 어떤 이벤트로 고생하고 있었는지 집계합니다.

```sql
SELECT
    event,           -- 발생한 대기 이벤트
    session_id,      -- 고통받고 있는 세션 ID (SID)
    COUNT(*) as wait_count -- 샘플링된 횟수 (이 숫자가 크면 오래 멈춰있었다는 뜻)
FROM v$active_session_history
WHERE sample_time > sysdate - 5/1440 -- 현재 시간 기준 최근 5분간의 기록
  AND event IS NOT NULL              -- 이벤트가 발생한 경우만 필터링
GROUP BY event, session_id
ORDER BY wait_count DESC; -- 많이 찍힌 순서대로 정렬
```

---

## 3. 범인(세션)의 '신상 정보' 및 '범행 도구' 확인
특정 세션 ID(SID)가 어떤 사용자이며, 현재 어떤 쿼리를 실행 중인지 상세히 조회합니다.

```sql
SELECT 
    s.sid, 
    s.serial#, 
    s.username, 
    s.osuser, 
    s.program, 
    q.sql_text -- 세션이 실행 중인 실제 SQL 문장
FROM v$session s
JOIN v$sql q ON s.sql_id = q.sql_id -- 현재 세션의 SQL ID를 기준으로 SQL 텍스트 결합
WHERE s.sid = 29; -- 조사 대상 세션 ID 입력
```

---

## 4. '계급도' 분석: 대기 세션 간의 연결 고리 (Hierarchy)
누가 누구를 붙잡고 있는지, 근본적인 원인을 제공하는 '최상위 블로킹 세션'을 계층형 쿼리로 찾습니다.

```sql
SELECT
    level, -- 계층 깊이 (1이 최상위 가해자)
    lpad(' ', (level-1)*2) || sid as "Session", -- 계층별 들여쓰기 시각화
    blocking_session as "Blocking", -- 나를 막고 있는 세션 ID
    event, -- 현재 대기 중인 이벤트
    seconds_in_wait -- 대기 중인 시간 (초)
FROM v$session
WHERE blocking_session IS NOT NULL -- 누군가를 막거나 막히고 있는 세션만 추출
   OR sid IN (SELECT blocking_session FROM v$session)
CONNECT BY PRIOR sid = blocking_session -- 내(sid)가 누군가의 블로킹 세션인 경우 연결
START WITH blocking_session IS NULL;    -- 나를 막는 놈이 없는 '최상위 가해자'부터 시작
```

SELECT tablespace_name,
       ROUND(SUM(bytes) / 1024 / 1024, 2) AS size_mb,
       ROUND(SUM(bytes) / 1024 / 1024 / 1024, 2) AS size_gb,
       ROUND(SUM(CASE WHEN status = 'ACTIVE'    THEN bytes ELSE 0 END) / 1024 / 1024, 2) AS active_mb,
       ROUND(SUM(CASE WHEN status = 'UNEXPIRED' THEN bytes ELSE 0 END) / 1024 / 1024, 2) AS unexpired_mb,
       ROUND(SUM(CASE WHEN status = 'EXPIRED'   THEN bytes ELSE 0 END) / 1024 / 1024, 2) AS expired_mb
FROM   dba_undo_extents
GROUP  BY tablespace_name;

SELECT s.sid,
       s.serial#,
       p.spid AS os_process_id, -- 이게 파일명 뒤에 붙는 번호입니다.
       s.username,
       s.program
FROM v$session s, v$process p
WHERE s.paddr = p.addr
  AND s.sid = (SELECT sid FROM v$mystat WHERE rownum = 1);
  
  
# ================================================
# 현재 오랜 시간 실행중인 sql 확인
# ================================================= 
  
-- 1. 화면 출력 가로 길이 설정
SET LINESIZE 200
SET PAGESIZE 50

-- 2. 컬럼별 포맷 설정
COL status         FOR a10      HEAD "Status"
COL username       FOR a12      HEAD "User"
COL sql_id         FOR a15      HEAD "SQL ID"
COL sql_exec_start FOR a20      HEAD "Start Time"
COL elapsed_sec    FOR 999,990.99 HEAD "Elapsed(s)"
COL sql_text       FOR a50      HEAD "SQL Text"

-- 3. 쿼리 실행
SELECT status, 
       username,
       sql_id, 
       TO_CHAR(sql_exec_start, 'YYYY/MM/DD HH24:MI:SS') as sql_exec_start,
       ROUND(elapsed_time / 1000000, 2) AS elapsed_sec, 
       SUBSTR(sql_text, 1, 50) as sql_text
FROM v$sql_monitor
WHERE status = 'EXECUTING';
