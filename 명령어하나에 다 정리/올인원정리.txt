# 컬럼별 너비 및 헤더 설정
COL file_id         FOR 999          HEAD 'ID'
COL tablespace_name FOR A20          HEAD 'Tablespace'
COL file_name       FOR A60          HEAD 'Physical_Path'
COL "Size(MB)"      FOR 9,999,999    HEAD 'Size(MB)'
COL "MaxSize(MB)"   FOR 9,999,999    HEAD 'Max(MB)'
COL "Auto"          FOR A5           HEAD 'Auto'
COL "Total(MB)"     FOR 9,999,999    HEAD 'Total(MB)'
COL "Used(MB)"      FOR 9,999,999    HEAD 'Used(MB)'
COL "Used(%)"       FOR 999.99       HEAD 'Used(%)'

### [조회 1] 모든 데이터파일의 실제 경로 및 용량
SELECT 
    file_id,
    tablespace_name,
    file_name,
    bytes/1024/1024 AS "Size(MB)",
    autoextensible AS "Auto"
FROM dba_data_files
ORDER BY tablespace_name, file_id;

### [조회 2] 임시(Temp) 파일 경로 확인
SELECT 
    file_id,
    tablespace_name,
    file_name,
    bytes/1024/1024 AS "Size(MB)"
FROM dba_temp_files;

### [조회 3] 테이블스페이스별 현재 사용량 (%)
SELECT 
    df.tablespace_name,
    ROUND(df.bytes / (1024 * 1024)) AS "Total(MB)",
    ROUND((df.bytes - NVL(SUM(fs.bytes),0)) / (1024 * 1024)) AS "Used(MB)",
    ROUND((df.bytes - NVL(SUM(fs.bytes),0)) / df.bytes * 100, 2) AS "Used(%)"
FROM dba_data_files df
LEFT JOIN dba_free_space fs ON df.file_id = fs.file_id
GROUP BY df.tablespace_name, df.bytes
ORDER BY 4 DESC;

# 📦 10. Data Pump 실행 옵션 (Practical Options)
# ============================================================

### [EXPORT] 자주 쓰는 내보내기 옵션 (expdp)
# - SCHEMAS: 특정 유저 전체 백업
# - PARALLEL: 프로세스 병렬 처리 (속도 향상)
# - COMPRESSION: 덤프 파일 압축 (용량 절감)
expdp hr/password@orcl \
DIRECTORY=dpump_dir \
DUMPFILE=hr_full_%U.dmp \
LOGFILE=exp_hr.log \
SCHEMAS=hr \
PARALLEL=4 \
COMPRESSION=ALL;

### [IMPORT] 자주 쓰는 들여오기 옵션 (impdp)
# - REMAP_SCHEMA: A유저 데이터를 B유저로 옮길 때
# - REMAP_TABLESPACE: 소스 테이블스페이스가 타겟에 없을 때
# - TABLE_EXISTS_ACTION: 테이블이 이미 있을 때 처리 (APPEND, REPLACE, TRUNCATE)
impdp system/password@orcl \
DIRECTORY=dpump_dir \
DUMPFILE=hr_full_%U.dmp \
LOGFILE=imp_hr.log \
PARALLEL=4 \
REMAP_SCHEMA=hr:scott \
REMAP_TABLESPACE=users:ts_data \
TABLE_EXISTS_ACTION=REPLACE;

### [FILTER] 특정 조건만 골라내기 (Include/Exclude)
# - EXCLUDE: 특정 오브젝트 제외 (예: 통계정보, 인덱스 제외)
# - CONTENT: 데이터만(DATA_ONLY) 또는 구조만(METADATA_ONLY)
expdp hr/password@orcl \
DIRECTORY=dpump_dir \
DUMPFILE=hr_meta.dmp \
CONTENT=METADATA_ONLY \
EXCLUDE=STATISTICS,INDEX;

### [권한] DB Link 생성을 위한 권한 부여 (System 계정에서 실행)
GRANT CREATE DATABASE LINK TO hr;
# (Public 생성 시) GRANT CREATE PUBLIC DATABASE LINK TO hr;

### [생성] 방법 1. TNS 정보를 직접 입력하는 방식 (권장)
# - tnsnames.ora 설정을 건드리지 않아도 되어 간편합니다.
CREATE DATABASE LINK remote_link
CONNECT TO remote_user IDENTIFIED BY "password123!"
USING '(DESCRIPTION=(ADDRESS=(PROTOCOL=TCP)(HOST=192.168.0.100)(PORT=1521))(CONNECT_DATA=(SID=orcl)))';

### [생성] 방법 2. TNS 명칭을 사용하는 방식
# - 로컬 서버의 tnsnames.ora에 'PROD_DB'가 정의되어 있어야 합니다.
CREATE DATABASE LINK prod_link
CONNECT TO scott IDENTIFIED BY tiger
USING 'PROD_DB';

### [조회] 생성된 DB Link 목록 확인
COL owner FOR A10
COL db_link FOR A20
COL host FOR A30
SELECT owner, db_link, username, host FROM dba_db_links;

### [사용] 원격지 테이블 조회 및 데이터 복사

SELECT * FROM emp@remote_link;

# - 원격지 데이터를 로컬로 바로 복사 (CTAS)
CREATE TABLE local_emp_bak AS SELECT * FROM emp@remote_link;

# ⚡ 16. 이미지 카피 - 데이터파일 단위
# ============================================================
RUN {
  # 1. 병렬 처리를 위해 채널 할당 (I/O 분산)
  ALLOCATE CHANNEL c1 DEVICE TYPE DISK;
  ALLOCATE CHANNEL c2 DEVICE TYPE DISK;

  # 2. 이미지 카피 생성 (핵심 옵션 조합)
  BACKUP AS COPY 
    DATAFILE 12                                      -- 대상: 데이터파일 12번 (ts700)
    FORMAT '/u01/app/oracle/oradata/ORCL/backup/df12_%U.copy' -- 저장 경로 및 포맷
    TAG 'IMG_COPY_TS700_LAB'                         -- 식별용 태그 이름
    CHECK LOGICAL                                    -- 내부 논리적 손상까지 검사
    SECTION SIZE 100M;                               -- 대용량 파일 병렬 카피 (멀티섹션)

  # 3. 생성된 카피본 정보 확인
  LIST COPY OF DATAFILE 12;

  RELEASE CHANNEL c1;
  RELEASE CHANNEL c2;
}

RUN {
  # 1. 채널 할당
  ALLOCATE CHANNEL c1 DEVICE TYPE DISK;
  ALLOCATE CHANNEL c2 DEVICE TYPE DISK;

  # 2. 전체 DB 이미지 카피 생성
  BACKUP AS COPY 
    DATABASE                                         -- 대상: 전체 데이터베이스
    FORMAT '/u01/app/oracle/oradata/ORCL/backup/db_full_%U.copy'
    TAG 'FULL_IMG_COPY_MAR_31'
    CHECK LOGICAL                                    -- 무결성 보장 옵션
    SECTION SIZE 100M;                               -- 전체 파일 병렬 처리 최적화

  # 3. 제어 파일 및 SPFILE 수동 카피 (선택 사항)
  BACKUP AS COPY CURRENT CONTROLFILE;
  BACKUP AS COPY SPFILE;

  RELEASE CHANNEL c1;
  RELEASE CHANNEL c2;
}

# ⚡ 19. INDEX REBUILD
# ============================================================

#일반
ALTER INDEX "HR"."DEPT_ID_PK" REBUILD ONLINE;

#병렬
ALTER INDEX "HR"."DEPT_ID_PK" REBUILD PARALLEL 4;
-- 작업 완료 후 다시 단일 프로세스 모드로 돌려주는 것이 좋습니다.
ALTER INDEX "HR"."DEPT_ID_PK" NOPARALLEL;

#모든 Unusable 인덱스 한 번에 리빌드
SELECT 'ALTER INDEX ' || owner || '.' || index_name || ' REBUILD;'
FROM dba_indexes
WHERE status = 'UNUSABLE' AND owner = 'HR';
