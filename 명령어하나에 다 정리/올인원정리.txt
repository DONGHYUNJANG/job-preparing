# ============================================================
# 경로 및 상태 조회 (Path & Status Inquiry)
# ============================================================

# 화면 설정: 한 줄 길이와 페이지당 행 수 최적화
SET LINESIZE 200
SET PAGESIZE 100
SET FEEDBACK OFF

# 컬럼별 너비 및 헤더 설정
COL file_id         FOR 999          HEAD 'ID'
COL tablespace_name FOR A20          HEAD 'Tablespace'
COL file_name       FOR A60          HEAD 'Physical_Path'
COL "Size(MB)"      FOR 9,999,999    HEAD 'Size(MB)'
COL "MaxSize(MB)"   FOR 9,999,999    HEAD 'Max(MB)'
COL "Auto"          FOR A5           HEAD 'Auto'
COL "Total(MB)"     FOR 9,999,999    HEAD 'Total(MB)'
COL "Used(MB)"      FOR 9,999,999    HEAD 'Used(MB)'
COL "Used(%)"       FOR 999.99       HEAD 'Used(%)'

### [조회 1] 모든 데이터파일의 실제 경로 및 용량
SELECT 
    file_id,
    tablespace_name,
    file_name,
    bytes/1024/1024 AS "Size(MB)",
    autoextensible AS "Auto"
FROM dba_data_files
ORDER BY tablespace_name, file_id;

### [조회 2] 임시(Temp) 파일 경로 확인
SELECT 
    file_id,
    tablespace_name,
    file_name,
    bytes/1024/1024 AS "Size(MB)"
FROM dba_temp_files;

### [조회 3] 테이블스페이스별 현재 사용량 (%)
SELECT 
    df.tablespace_name,
    ROUND(df.bytes / (1024 * 1024)) AS "Total(MB)",
    ROUND((df.bytes - NVL(SUM(fs.bytes),0)) / (1024 * 1024)) AS "Used(MB)",
    ROUND((df.bytes - NVL(SUM(fs.bytes),0)) / df.bytes * 100, 2) AS "Used(%)"
FROM dba_data_files df
LEFT JOIN dba_free_space fs ON df.file_id = fs.file_id
GROUP BY df.tablespace_name, df.bytes
ORDER BY 4 DESC;

### [조회] 현재 내 세션의 NLS 설정 확인
# - 어떤 포맷이 적용되어 있는지 먼저 확인합니다.

FROM v$nls_parameters 
WHERE parameter IN ('NLS_DATE_FORMAT', 'NLS_LANGUAGE', 'NLS_TERRITORY', 'NLS_TIMESTAMP_FORMAT');

### [날짜] 날짜 출력 형식 변경 (Date Format)
# - 기본: RR/MM/DD -> 상세: YYYY-MM-DD HH24:MI:SS

# - 타임스탬프(밀리초 포함) 포맷 변경

### [조회] 패스워드 파일에 저장된 유저 확인
# - 어떤 유저가 관리자 권한을 가지고 파일에 등록되어 있는지 확인
COL username FOR A20
COL sysdba   FOR A10
SELECT username, sysdba, sysoper, sysbackup FROM v$pwfile_users;

### [확인] 패스워드 파일 설정 및 상태 확인
# - VALUE가 'EXCLUSIVE'여야 개별 유저 등록이 가능함 (기본값)
# - VALUE가 'NONE'이면 패스워드 파일 사용 안 함
SHOW PARAMETER remote_login_passwordfile;

### [등록/해제] 패스워드 파일에 유저 추가 및 삭제
# - 유저 등록 (SYSDBA 권한 부여 시 자동으로 파일에 기록됨)
GRANT SYSDBA TO hr;

# - 유저 해제 (권한 박탈 시 파일에서 자동 삭제됨)
REVOKE SYSDBA FROM hr;

### [생성/복구] 패스워드 파일 생성 (orapwd 유틸리티)
# - 파일이 삭제되었거나 비밀번호를 초기화해야 할 때 사용 (OS 커맨드)
# - entries: 등록 가능한 최대 유저 수
orapwd file=$ORACLE_HOME/dbs/orapw<SID> password=<sys_password> entries=10 force=y format=12;

### [재생성] 패스워드 파일 포맷 업그레이드 (12c 이상)
# - 구버전 포맷을 12c 이상의 새로운 포맷으로 변경할 때
orapwd file=$ORACLE_HOME/dbs/orapw<SID> dbname=<SID> format=12 force=y;
